# proactnaming_azurerm_resources (Data Source)

Returns a `map(string)` of azurerm Terraform resource types to their short names. The key is the azurerm resource type (e.g. `azurerm_storage_account`) and the value is the corresponding short name used by the naming tool (e.g. `st`).

Only resource types without a property qualifier are included (i.e. the same set as the `azurerm_resource_type` field from `proactnaming_resource_types` where `property == ""`).

## Example Usage

### Basic Lookup

```terraform
data "proactnaming_azurerm_resources" "mapping" {}

output "storage_short_name" {
  value = data.proactnaming_azurerm_resources.mapping.resources["azurerm_storage_account"]
  # => "st"
}
```

### Generate a Name from an azurerm Resource Type

```terraform
data "proactnaming_azurerm_resources" "mapping" {}

resource "proactnaming_generate_name" "storage" {
  organization  = "myorg"
  resource_type = data.proactnaming_azurerm_resources.mapping.resources["azurerm_storage_account"]
  application   = "webapp"
  function      = "data"
  instance      = "001"
  location      = "euw"
  environment   = "dev"
}

output "storage_name" {
  value = proactnaming_generate_name.storage.resource_name
}
```

### Generate Names for Multiple azurerm Resources

```terraform
data "proactnaming_azurerm_resources" "mapping" {}

locals {
  resources_to_name = [
    "azurerm_resource_group",
    "azurerm_storage_account",
    "azurerm_kubernetes_cluster",
    "azurerm_key_vault",
  ]
}

resource "proactnaming_generate_name" "infra" {
  for_each = toset(local.resources_to_name)

  organization  = "myorg"
  resource_type = data.proactnaming_azurerm_resources.mapping.resources[each.value]
  application   = "platform"
  function      = "core"
  instance      = "001"
  location      = "euw"
  environment   = "prod"
}

output "generated_names" {
  value = { for k, v in proactnaming_generate_name.infra : k => v.resource_name }
}
```

data "proactnaming_azurerm_resources" "mapping" {}
