# proactnaming_azurerm_resources (Data Source)

Returns a map of azurerm Terraform resource types to their short names. Use this data source to easily look up the short name for a given azurerm resource type (e.g. `azurerm_storage_account` => `st`). Only resource types without a property qualifier are included.

## Example Usage

### Basic Usage

```terraform
# Fetch the complete map of azurerm resource types to short names
data "proactnaming_azurerm_resources" "mapping" {}

output "all_mappings" {
  description = "Full map of azurerm resource type to short name."
  value       = data.proactnaming_azurerm_resources.mapping.resources
}
```

### Look Up a Short Name Dynamically

Use this data source together with `proactnaming_generate_name` so you don't need to hardcode short names (e.g. `"st"`) in your configuration — the provider derives them from your naming tool automatically.

```terraform
data "proactnaming_azurerm_resources" "mapping" {}

locals {
  # Resolve short names for the resource types we need.
  rg_short  = data.proactnaming_azurerm_resources.mapping.resources["azurerm_resource_group"]
  st_short  = data.proactnaming_azurerm_resources.mapping.resources["azurerm_storage_account"]
  aks_short = data.proactnaming_azurerm_resources.mapping.resources["azurerm_kubernetes_cluster"]
}

resource "proactnaming_generate_name" "rg" {
  organization  = "myorg"
  resource_type = local.rg_short
  application   = "webapp"
  instance      = "001"
  location      = "euw"
  environment   = "prod"
}

resource "proactnaming_generate_name" "storage" {
  organization  = "myorg"
  resource_type = local.st_short
  application   = "webapp"
  function      = "data"
  instance      = "001"
  location      = "euw"
  environment   = "prod"
}
```

### Validate That Required Types Exist

```terraform
data "proactnaming_azurerm_resources" "mapping" {}

locals {
  required_types = [
    "azurerm_resource_group",
    "azurerm_storage_account",
    "azurerm_kubernetes_cluster",
  ]

  missing_types = [
    for t in local.required_types : t
    if !contains(keys(data.proactnaming_azurerm_resources.mapping.resources), t)
  ]
}

# This will fail the plan if any required type is missing from the naming tool.
resource "terraform_data" "assert_types_present" {
  lifecycle {
    precondition {
      condition     = length(local.missing_types) == 0
      error_message = "The following azurerm types are not mapped in the naming tool: ${join(", ", local.missing_types)}"
    }
  }
}
```
